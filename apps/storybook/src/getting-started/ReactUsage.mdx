<div className="sb-unstyled">
  <div className="docs-section" style={{ marginTop: '48px' }}>
    <div className="docs-section-title">Import Paths</div>
    <div className="docs-subsection-description">
      Components can be imported from either the root or the <code>/react</code> subpath. Both
      exports are identical.
    </div>
  </div>
</div>

```tsx
// Root import (default)
import { Button, InputField, SegmentedControl } from '@ckw-elements/components';

// Explicit React subpath (identical)
import { Button, InputField, SegmentedControl } from '@ckw-elements/components/react';
```

```tsx
import React from 'react';
import { Button } from '@ckw-elements/components/react';

function MyPage() {
  return (
    <div>
      <Button type="Primary" size="Large" onClick={() => alert('clicked')}>
        Submit
      </Button>
      <Button type="Secondary" size="Medium">
        Cancel
      </Button>
      <Button type="Tertiary" size="Small">
        Learn more
      </Button>
    </div>
  );
}
```
