<div className="sb-unstyled">
  <div className="docs-section" style={{ marginTop: '48px' }}>
    <div className="docs-section-title">Import Paths</div>
    <div className="docs-subsection-description">
      Components can be imported from either the root or the <code>/react</code> subpath. Both
      exports are identical.
    </div>
  </div>
</div>

```tsx
// Root import (default)
import { Button, InlineMessage, InputField, SegmentedControl } from '@ckw-elements/components';
import { IconSearch } from '@ckw-elements/icons';

// Explicit React subpath (identical)
import {
  Button,
  InlineMessage,
  InputField,
  SegmentedControl,
} from '@ckw-elements/components/react';
```

<div className="sb-unstyled">

    <div className="docs-section-title">Token Setup</div>
    <div className="docs-subsection-description">
      Import the token CSS file once in your app entry point. Components rely on CSS custom
      properties being available in the document. Component CSS is imported by the component
      package; import the aggregate stylesheet only if your bundler does not process package CSS
      side effects.
    </div>
  </div>
</div>

```css
/* app entry point (e.g., main.css, global.css, or index.css) */
@import '@ckw-elements/tokens/tokens.css';

/* optional fallback for bundlers that ignore package CSS side effects */
@import '@ckw-elements/components/styles.css';
```

    <div className="docs-subsection-description">

```tsx
import React from 'react';
import { Button } from '@ckw-elements/components/react';
import { IconArrowRight, IconSearch } from '@ckw-elements/icons/react';

function MyPage() {
  return (
    <div>
      <InlineMessage status="Info" title="Statusmeldung">
        Hier steht eine kurze Erklaerung zum aktuellen Status oder zur naechsten Aktion.
      </InlineMessage>
      <InputField label="Search" icon={<IconSearch size={20} />} />
      <Button
        type="Primary"
        size="Large"
        trailingIcon={<IconArrowRight />}
        onClick={() => alert('clicked')}
      >
        Submit
      </Button>
      <Button type="Secondary" size="Medium">
        Cancel
      </Button>
      <Button type="Tertiary" size="Small">
        Learn more
      </Button>
    </div>
  );
}
```
