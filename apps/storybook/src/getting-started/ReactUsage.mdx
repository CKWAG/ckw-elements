    <div className="docs-section-title">Token Setup</div>
    <div className="docs-subsection-description">
      Import the token CSS file once before rendering CKW components. Component CSS is imported by
      the component package; import the aggregate stylesheet explicitly when your bundler does not
      process package CSS side effects or when you want one predictable global stylesheet import.
    </div>
  </div>

```css
/* app entry point (e.g., main.css, global.css, or index.css) */
@import '@ckwag/elements-tokens/tokens.css';

/* optional fallback for bundlers that ignore package CSS side effects */
@import '@ckwag/elements/styles.css';
```

<div className="sb-unstyled">

      The default imports above preserve existing cascade behavior. For applications that use CSS

```css
@layer ckw-tokens, ckw-elements, app;

@import '@ckwag/elements-tokens/tokens.layered.css';
@import '@ckwag/elements/styles.layered.css';

@layer app {
  .page {
    color: var(--text-primary);
  }
}
```

<div className="sb-unstyled">
  <div className="docs-section" style={{ marginTop: '48px' }}>
    <div className="docs-section-title">Next.js App Router</div>
    <div className="docs-subsection-description">
      In Next.js App Router, import global CSS from <code>app/layout.tsx</code>. If you use
      <code>next/font/local</code>, expose Gotham on a separate variable and bridge it back to the
      CKW token in your global CSS. This keeps CKW components coupled to the design token, not to
      Next.js internals.
    </div>
  </div>
</div>

```tsx
// app/fonts.ts
import localFont from 'next/font/local';

export const gotham = localFont({
  src: [
    { path: '../public/fonts/Gotham-Book.woff2', weight: '325', style: 'normal' },
    { path: '../public/fonts/Gotham-Medium.woff2', weight: '500', style: 'normal' },
    { path: '../public/fonts/Gotham-Bold.woff', weight: '700', style: 'normal' },
  ],
  variable: '--font-gotham',
  display: 'swap',
  fallback: ['Helvetica Neue', 'Arial', 'system-ui', 'sans-serif'],
});
```

```tsx
// app/layout.tsx
import React from 'react';
import '@ckwag/elements-tokens/tokens.css';
import '@ckwag/elements/styles.css';
import './globals.css';
import { gotham } from './fonts';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="de" className={gotham.variable}>
      <body>{children}</body>
    </html>
  );
}
```

```css
/* app/globals.css — imported after CKW tokens in app/layout.tsx */
:root {
  --font-family-brand: var(--font-gotham, Gotham, 'Helvetica Neue', Arial, system-ui, sans-serif);
}
```

    <div className="docs-subsection-description">

      InlineMessage are presentational components. InputField, PasswordInput, and SegmentedControl

      interactivity. Props extend native HTML props where practical: Button extends native button
      props, InputField extends native input props, and PasswordInput composes InputField while
      omitting the internally controlled <code>type</code> and <code>icon</code> props.

```tsx
import React from 'react';
import { Button, InlineMessage, InputField } from '@ckwag/elements';
import { IconArrowRight, IconSearch } from '@ckwag/elements-icons';

function MyPage() {
  return (
    <div>
      <InlineMessage status="Info" title="Statusmeldung">
        Hier steht eine kurze Erklaerung zum aktuellen Status oder zur naechsten Aktion.
      </InlineMessage>
      <InputField label="Search" icon={<IconSearch size={20} />} />
      <Button
        variant="primary"
        size="Large"
        trailingIcon={<IconArrowRight />}
        onClick={() => alert('clicked')}
      >
        Submit
      </Button>
      <Button variant="secondary" size="Medium">
        Cancel
      </Button>
      <Button variant="tertiary" size="Small">
        Learn more
      </Button>
    </div>
  );
}
```
