```html
<!-- HTML: default variant (light backgrounds) -->
<img src="https://ckwag.github.io/ckw-elements/logo-default.svg" alt="CKW" />

<!-- HTML: inverted variant (dark backgrounds) -->
<img src="https://ckwag.github.io/ckw-elements/logo-inverted.svg" alt="CKW" />
```

```tsx
// React: import as asset (Vite/webpack)
import React from 'react';
import ckwLogoDefault from './assets/logo-default.svg';
import ckwLogoInverted from './assets/logo-inverted.svg';

export function Header({ variant = 'default' }) {
  const src = variant === 'inverted' ? ckwLogoInverted : ckwLogoDefault;
  return (
    <header>
      <img src={src} alt="CKW" width={120} />
    </header>
  );
}
```

```css
/* CSS: as background image */
.brand-logo {
  width: 160px;
  height: 99px;
  background: url('/logo-default.svg') no-repeat center;
  background-size: contain;
}

/* Dark section */
.dark-section .brand-logo {
  background-image: url('/logo-inverted.svg');
}
```
